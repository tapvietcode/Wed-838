import pygame, random, sys, time pygame.init() W, H = 800, 600 screen = pygame.display.set_mode((W, H)) pygame.display.set_caption("Aim Trainer - Circle") clock = pygame.time.Clock() font = pygame.font.SysFont(None, 36) def new_target(): r = random.randint(20, 45) return [random.randint(r, W - r), random.randint(r + 60, H - r), r] target = new_target() score = 0 misses = 0 start = time.time() duration = 60 running = True while running: clock.tick(120) for e in pygame.event.get(): if e.type == pygame.QUIT: running = False elif e.type == pygame.MOUSEBUTTONDOWN and e.button == 1: mx, my = e.pos tx, ty, tr = target if (mx - tx) ** 2 + (my - ty) ** 2 <= tr ** 2: score += 1 target = new_target() else: misses += 1 elapsed = time.time() - start if elapsed >= duration: running = False screen.fill((20, 20, 25)) tx, ty, tr = target pygame.draw.circle(screen, (0, 220, 140), (tx, ty), tr) pygame.draw.circle(screen, (255, 255, 255), (tx, ty), tr, 2) hud = font.render( f"Score: {score} Miss: {misses} Time: {max(0, duration - elapsed):.1f}", True, (230, 230, 230) ) screen.blit(hud, (10, 10)) pygame.display.flip() pygame.quit() sys.exit()
